#include "IPBoundary.h"
#include "llvm/Support/raw_ostream.h"
using namespace llvm;
using namespace llvm::sampleprof;
int main(int argc, char **argv) {
  if (argc != 2) return 1;
  auto M = IPBoundaryMap::load(argv[1], "");
  if (!M) { logAllUnhandledErrors(M.takeError(), errs()); return 1; }
  std::pair<uint64_t,uint64_t> Range{0x1000,0x1004};
  auto Text=M->enumerate("text", {Range}, ".text");
  auto Plt=M->enumerate("plt", {Range}, ".plt");
  if (!Text) { logAllUnhandledErrors(Text.takeError(), errs()); return 1; }
  if (!Plt) { logAllUnhandledErrors(Plt.takeError(), errs()); return 1; }
  Text->print(outs()); Plt->print(outs());
  if (Text->Instructions.size()!=2 || Text->Instructions[0].Address!=0x1000 ||
      Text->Instructions[1].Address!=0x1002 || !Plt->Instructions.empty()) return 1;
  auto Ambiguous=M->enumerate("ambiguous", {Range});
  if (Ambiguous) return 1;
  std::string E=toString(Ambiguous.takeError());
  if (E!="ambiguous section for DWARF range") return 1;
  outs()<<"PASS section-local state; unqualified query rejected: "<<E<<'\n';
}
