#!/bin/bash
set -Eeuo pipefail
W=/home/linhao/Toolchain/development/llvm_autofdo_bolt_agent
export TMPDIR="$W/temp/tmp"
run=${1:?run label required}
[[ $run =~ ^[a-z0-9_-]+$ ]] || exit 2
shift
if [[ ${1:-} != --worker ]]; then
    test ! -e /etc/gbs.conf
    test -f "$W/temp/gbs-home/.gbs.conf"
    test -z "$(git -C "$W/temp/rpi4-followup/spike-arm" status --porcelain)"
    exec 9>"$W/temp/rpi4-followup/build.lock"
    flock -n 9 || { echo 'Another spike build is running'; exit 1; }
    setsid --fork /bin/bash "$W/temp/rpi4-followup/build.sh" "$run" --worker "$@" 9>&9 >"$W/temp/rpi4-followup/logs/$run.log" 2>&1 </dev/null
    echo "Worker launched: $W/temp/rpi4-followup/logs/$run.status"
    exit 0
fi
shift
status="$W/temp/rpi4-followup/logs/$run.status"
start=$(date --iso-8601=seconds)
rc=NOT_STARTED
finish() {
    code=$?
    trap - EXIT
    set +e
    end=$(date --iso-8601=seconds)
    verdict=FAILED
    [[ $rc == 0 && $code == 0 ]] && verdict=OK
    {
        printf 'START %s\nEND %s\nGBS_EXIT %s\nWRAPPER_EXIT %s\n' "$start" "$end" "$rc" "$code"
        find "$W/temp/GBS-ROOT-RPI4-SPIKE/local/repos" -type f -name '*.rpm' -print
        printf 'ALL-END %s %s\n' "$end" "$verdict"
    } >"$status.tmp"
    mv "$status.tmp" "$status"
    cat "$status"
}
trap finish EXIT
trap '' HUP
trap 'exit 143' TERM
trap 'exit 130' INT
printf 'START %s\nPHASE BUILDING\nWORKER_PID %s\n' "$start" "$$" >"$status"
cd "$W/temp/rpi4-followup/spike-arm"
set +e
setsid --wait env HOME="$W/temp/gbs-home" GIT_CONFIG_NOSYSTEM=1 TMPDIR="$W/temp/gbs-tmp-rpi4-spike" nice -n 19 ionice -c3 gbs -c "$W/temp/gbs/gbs_rpi4_spike.conf" build -A armv7l --threads 1 --define '_smp_mflags -j2' "$@"
rc=$?
exit "$rc"
